## The parameters to tune, generated from NSGAIIDouble.yaml by
## org.uma.evolver.irace.generator.IraceNSGAIIDoubleParameterDescriptionGenerator.
parameterFile = "parameters-NSGAII.txt"

## Training instances: one line per instance, with the arguments the target runner receives for it.
trainInstancesDir = ""
trainInstancesFile = "instances-list-ZDT.txt"

## The target runner: irace runs "java -cp <jar> ..." for every experiment (a configuration on an
## instance), and reads the value it prints on its last line, which it minimizes.
targetRunner = "Evolver-jar-with-dependencies.jar"
targetRunnerLauncher = "java"
targetCmdline = "-cp {targetRunner} org.uma.evolver.irace.AutoNSGAIIIraceHVEP --randomGeneratorSeed {seed} --problemName {instance} {targetRunnerArgs}"

## Budget: the maximum number of experiments. irace needs at least 3640 with this parameter space
## and these settings, and stops with an "Insufficient budget" error below that.
maxExperiments = 10000

## Use the instances in the order of the file, in blocks of 5 (the five ZDT problems), so that
## configurations are always compared after running on whole passes over the training set. With
## irace's default firstTest (5 blocks) and eachTest (1 block), the first statistical test comes
## after 25 experiments per configuration, and a new one after each block.
sampleInstances = 0
blockSize = 5
